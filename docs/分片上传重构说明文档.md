## 核心算法实现

### 1. 动态分片大小调整算法

```typescript
/**
 * 动态计算最优分片大小
 * 考虑因素：文件大小、网络状态、历史性能
 */
private calculateOptimalChunkSize(
  fileSize: number, 
  networkStatus: NetworkStatus,
  averageSpeed?: number
): number {
  let chunkSize = CONFIG.CHUNK_SIZE.DEFAULT;
  
  // 基于文件大小的基础调整
  if (fileSize < 50 * 1024 * 1024) {
    chunkSize = CONFIG.CHUNK_SIZE.MIN;  // 小文件使用小分片
  } else if (fileSize > 500 * 1024 * 1024) {
    chunkSize = CONFIG.CHUNK_SIZE.MAX;  // 大文件使用大分片
  }
  
  // 基于网络状态的动态调整
  switch (networkStatus) {
    case NetworkStatus.FAST:
      chunkSize = Math.min(chunkSize * 2, CONFIG.CHUNK_SIZE.MAX);
      break;
    case NetworkStatus.SLOW:
      chunkSize = Math.max(chunkSize / 2, CONFIG.CHUNK_SIZE.MIN);
      break;
  }
  
  // 基于历史速度的微调
  if (averageSpeed) {
    if (averageSpeed > CONFIG.NETWORK.FAST_THRESHOLD) {
      chunkSize = Math.min(chunkSize * 1.5, CONFIG.CHUNK_SIZE.MAX);
    } else if (averageSpeed < CONFIG.NETWORK.SLOW_THRESHOLD) {
      chunkSize = Math.max(chunkSize * 0.7, CONFIG.CHUNK_SIZE.MIN);
    }
  }
  
  return Math.floor(chunkSize);
}
```

**算法特点**：
- **多因素考虑**：文件大小、网络状态、历史性能
- **边界保护**：确保分片大小在合理范围内
- **平滑调整**：避免剧烈的参数变化

```typescript
private calculateOptimalConcurrency(

  return concurrency;
}
```

```typescript
/**
 * 计算重试延迟（指数退避）
 * 避免重试风暴，提高成功率

private calculateRetryDelay(retryCount: number): number {
  const delay = CONFIG.RETRY.BASE_DELAY * 
    Math.pow(CONFIG.RETRY.BACKOFF_FACTOR, retryCount);
  return Math.min(delay, CONFIG.RETRY.MAX_DELAY);
}
```

## 性能优化策略

### 1. 内存优化

#### 问题分析
- 原有方案：所有分片同时加载到内存
- 内存占用：大文件可能占用数百MB内存
- 性能影响：可能导致浏览器卡顿

#### 优化方案
```typescript
/**
 * 按需创建分片，避免内存占用过大
 */
private createDynamicChunks(file: File, chunkSize: number) {
  const chunks: Blob[] = [];
  const chunkStates: ChunkState[] = [];
  
  // 只创建分片元数据，不立即切片
  let start = 0;
  let index = 0;
  
  while (start < file.size) {
    const end = Math.min(start + chunkSize, file.size);
    
    // 延迟创建实际分片数据
    chunks.push(file.slice(start, end));
    chunkStates.push({
      index,
      size: end - start,
      uploaded: false,
      uploading: false,
      retryCount: 0
    });
    
    start = end;
    index++;
  }
  
  return { chunks, chunkStates };
}
```

#### 优化效果
- **内存占用减少**：按需创建分片
- **响应性提升**：避免大文件切片阻塞UI
- **稳定性增强**：减少内存不足导致的崩溃

#### 智能队列管理
```typescript
/**
 * 并行上传分片，智能控制并发数
 */
private async uploadChunksInParallel(
  chunks: Blob[],
  fileId: string,
  state: FileUploadState,
  callbacks: ProgressCallbacks
): Promise<void> {
  const pendingChunks = state.chunks.filter(c => !c.uploaded && !c.uploading);
  
  while (pendingChunks.length > 0 && !state.isCancelled) {
    // 动态获取当前最优并发数
    const concurrency = Math.min(
      state.stats.currentConcurrency, 
      pendingChunks.length
    );
    
    // 选择要上传的分片
    const chunksToUpload = pendingChunks.splice(0, concurrency);
    
    // 并行上传当前批次
    const uploadPromises = chunksToUpload.map(chunkState => 
      this.uploadChunkWithRetry(chunks[chunkState.index], chunkState, fileId, state.uploadId!, callbacks)
    );
    
    // 等待当前批次完成
    await Promise.allSettled(uploadPromises);
    
    // 动态调整参数
    this.adjustUploadParameters(fileId, state);
  }
}
```

#### 负载均衡
- **动态调整**：根据网络状态和错误率调整并发数

### 2. 恢复策略

#### 智能重试
```typescript
private async uploadChunkWithRetry(
  chunk: Blob,
  chunkState: ChunkState,
  fileId: string,
  uploadId: string,
  callbacks: ProgressCallbacks
): Promise<void> {
  while (chunkState.retryCount < CONFIG.RETRY.MAX_ATTEMPTS) {
    try {
      // 执行上传
      await this.performChunkUpload(chunk, chunkState, uploadId);
      return; // 成功，退出重试循环
      
    } catch (error: any) {
      chunkState.retryCount++;
      
      // 处理不同类型的错误
      if (this.isRetryableError(error)) {
        const delay = this.calculateRetryDelay(chunkState.retryCount - 1);
        await new Promise(resolve => setTimeout(resolve, delay));
      } else {
        throw error; // 不可重试的错误直接抛出
      }
    }
  }
  
  throw new Error(`分片 ${chunkState.index + 1} 重试次数用完`);
}
```

```typescript
/**
 * 更新上传统计信息
 * 使用移动平均算法平滑速度计算
 */
private updateStats(fileId: string, state: FileUploadState): void {
  // 计算实时速度
  const completedChunks = state.chunks.filter(c => c.uploaded && c.startTime && c.endTime);
  
  if (completedChunks.length > 0) {
    // 计算每个分片的速度
    const speeds = completedChunks.map(c => {
      const time = (c.endTime! - c.startTime!) / 1000;
      return c.size / time;
    });
    
    // 使用移动平均算法
    state.speedHistory.push(...speeds);
    if (state.speedHistory.length > CONFIG.PERFORMANCE.SPEED_SAMPLE_SIZE) {
      state.speedHistory = state.speedHistory.slice(-CONFIG.PERFORMANCE.SPEED_SAMPLE_SIZE);
    }
    
    // 计算平滑平均速度
    const averageSpeed = state.speedHistory.reduce((sum, speed) => sum + speed, 0) / state.speedHistory.length;
    
    // 估算剩余时间
    const remainingBytes = state.stats.totalBytes - state.stats.uploadedBytes;
    const estimatedTimeRemaining = averageSpeed > 0 ? remainingBytes / averageSpeed : 0;
    
    // 更新统计信息
    state.stats.averageSpeed = averageSpeed;
    state.stats.estimatedTimeRemaining = estimatedTimeRemaining;
  }
}
```
