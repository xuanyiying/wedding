# 婚礼主持俱乐部官方网站 - 开发环境搭建

## 🛠️ 开发环境概述

### 开发目标
- **快速启动**：一键搭建完整开发环境
- **热重载**：代码修改实时生效，提高开发效率
- **调试友好**：完善的调试工具和日志输出
- **环境一致**：开发环境与生产环境保持一致

### 技术栈
- **前端**：React 18 + TypeScript + Vite + Ant Design
- **后端**：Node.js + Express + TypeScript + Prisma
- **数据库**：MySQL 8.0 + Redis 7.0
- **容器化**：Docker + Docker Compose
- **代码质量**：ESLint + Prettier + Husky

      dockerfile: Dockerfile.dev

      context: ./server
      dockerfile: Dockerfile.dev
    volumes:
      - ./server/src:/app/src
      - ./server/uploads:/app/uploads
    environment:
      - NODE_ENV=development
    command: npm run dev

  mysql:
    ports:
      - "3306:3306"
    volumes:
      - mysql_dev_data:/var/lib/mysql
      - ./deployment/docker/mysql/dev.cnf:/etc/mysql/conf.d/dev.cnf

  redis:
    ports:
      - "6379:6379"
    command: redis-server --appendonly yes --requirepass ${REDIS_PASSWORD}

  phpmyadmin:
    image: phpmyadmin/phpmyadmin:latest
    container_name: wedding-phpmyadmin-dev
    ports:
      - "8080:80"
    environment:
      PMA_HOST: mysql
      PMA_USER: root
      PMA_PASSWORD: ${DB_PASSWORD}
    depends_on:
      - mysql

volumes:
  mysql_dev_data:

```javascript
// 前端调试配置
// web/src/utils/debug.ts
export const isDev = import.meta.env.DEV;

export const logger = {
  debug: (...args: any[]) => {
    if (isDev) {
      console.log('[DEBUG]', ...args);
    }
  },
  info: (...args: any[]) => {
    if (isDev) {
      console.info('[INFO]', ...args);
    }
  },
  warn: (...args: any[]) => {
    console.warn('[WARN]', ...args);
  },
  error: (...args: any[]) => {
    console.error('[ERROR]', ...args);
  }
};

// 性能监控
export const performanceMonitor = {
  start: (label: string) => {
    if (isDev) {
      performance.mark(`${label}-start`);
    }
  },
  end: (label: string) => {
    if (isDev) {
      performance.mark(`${label}-end`);
      performance.measure(label, `${label}-start`, `${label}-end`);
      const measure = performance.getEntriesByName(label)[0];
      console.log(`[PERF] ${label}: ${measure.duration.toFixed(2)}ms`);
    }
  }
};
```

### API调试
```javascript
// 后端调试配置
// server/src/utils/logger.ts
import winston from 'winston';

const logger = winston.createLogger({
  level: process.env.LOG_LEVEL || 'info',
  format: winston.format.combine(
    winston.format.timestamp(),
    winston.format.errors({ stack: true }),
    winston.format.json()
  ),
  defaultMeta: { service: 'wedding-api' },
  transports: [
    new winston.transports.File({ filename: 'logs/error.log', level: 'error' }),
    new winston.transports.File({ filename: 'logs/combined.log' })
  ]
});

if (process.env.NODE_ENV !== 'production') {
  logger.add(new winston.transports.Console({
    format: winston.format.combine(
      winston.format.colorize(),
      winston.format.simple()
    )
  }));
}

export default logger;

// 请求日志中间件
// server/src/middlewares/requestLogger.ts
import { Request, Response, NextFunction } from 'express';
import logger from '../utils/logger';

export const requestLogger = (req: Request, res: Response, next: NextFunction) => {
  const start = Date.now();
  
  res.on('finish', () => {
    const duration = Date.now() - start;
    logger.info('HTTP Request', {
      method: req.method,
      url: req.url,
      status: res.statusCode,
      duration: `${duration}ms`,
      userAgent: req.get('User-Agent'),
      ip: req.ip
    });
  });
  
  next();
};
```
