## 🚀 分片上传实现

### 前端分片上传
```typescript
interface ChunkInfo {
  chunkIndex: number;
  chunkSize: number;
  totalChunks: number;
  fileName: string;
  fileHash: string;
}

class ChunkUploader {
  private chunkSize = 2 * 1024 * 1024; // 2MB per chunk
  
  /**
   * 分片上传文件
   */
  async uploadFile(file: File, onProgress?: (progress: number) => void): Promise<string> {
    // 1. 计算文件哈希
    const fileHash = await this.calculateFileHash(file);
    
    // 2. 检查文件是否已存在（秒传）
    const existingFile = await this.checkFileExists(fileHash);
    if (existingFile) {
      return existingFile.url;
    }
    
    // 3. 分片上传
    const totalChunks = Math.ceil(file.size / this.chunkSize);
    const uploadedChunks: number[] = [];
    
    for (let i = 0; i < totalChunks; i++) {
      const start = i * this.chunkSize;
      const end = Math.min(start + this.chunkSize, file.size);
      const chunk = file.slice(start, end);
      
      const chunkInfo: ChunkInfo = {
        chunkIndex: i,
        chunkSize: chunk.size,
        totalChunks,
        fileName: file.name,
        fileHash
      };
      
      await this.uploadChunk(chunk, chunkInfo);
      uploadedChunks.push(i);
      
      // 更新进度
      const progress = (uploadedChunks.length / totalChunks) * 100;
      onProgress?.(progress);
    }
    
    // 4. 合并分片
    const result = await this.mergeChunks(fileHash, file.name, totalChunks);
    return result.url;
  }
  
  /**
   * 计算文件哈希
   */
  private async calculateFileHash(file: File): Promise<string> {
    return new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onload = async (e) => {
        try {
          const buffer = e.target?.result as ArrayBuffer;
          const hashBuffer = await crypto.subtle.digest('SHA-256', buffer);
          const hashArray = Array.from(new Uint8Array(hashBuffer));
          const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
          resolve(hashHex);
        } catch (error) {
          reject(error);
        }
      };
      reader.onerror = reject;
      reader.readAsArrayBuffer(file);
    });
  }
  
  /**
   * 检查文件是否已存在
   */
  private async checkFileExists(fileHash: string): Promise<{ url: string } | null> {
    try {
      const response = await fetch(`/api/v1/files/check/${fileHash}`);
      if (response.ok) {
        return await response.json();
      }
      return null;
    } catch (error) {
      console.error('Check file exists error:', error);
      return null;
    }
  }
  
  /**
   * 上传单个分片
   */
  private async uploadChunk(chunk: Blob, chunkInfo: ChunkInfo): Promise<void> {
    const formData = new FormData();
    formData.append('chunk', chunk);
    formData.append('chunkIndex', chunkInfo.chunkIndex.toString());
    formData.append('totalChunks', chunkInfo.totalChunks.toString());
    formData.append('fileName', chunkInfo.fileName);
    formData.append('fileHash', chunkInfo.fileHash);
    
    const response = await fetch('/api/v1/files/upload-chunk', {
      method: 'POST',
      body: formData,
      headers: {
        'Authorization': `Bearer ${localStorage.getItem('accessToken')}`
      }
    });
    
    if (!response.ok) {
      throw new Error(`Chunk upload failed: ${response.statusText}`);
    }
  }
  
  /**
   * 合并分片
   */
  private async mergeChunks(fileHash: string, fileName: string, totalChunks: number): Promise<{ url: string }> {
    const response = await fetch('/api/v1/files/merge-chunks', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${localStorage.getItem('accessToken')}`
      },
      body: JSON.stringify({
        fileHash,
        fileName,
        totalChunks
      })
    });
    
    if (!response.ok) {
      throw new Error(`Merge chunks failed: ${response.statusText}`);
    }
    
    return await response.json();
  }
}
```

### 后端分片处理
```typescript
import { Request, Response } from 'express';
import { promises as fs } from 'fs';
import path from 'path';
import { v4 as uuidv4 } from 'uuid';

export class FileUploadController {
  private uploadDir = process.env.UPLOAD_PATH || '/app/uploads';
  private tempDir = path.join(this.uploadDir, 'temp');
  
  constructor() {
    this.ensureDirectories();
  }
  
  /**
   * 检查文件是否已存在
   */
  public checkFileExists = async (req: Request, res: Response): Promise<void> => {
    try {
      const { fileHash } = req.params;
      
      // 检查数据库中是否存在该文件
      const existingFile = await this.fileService.findByHash(fileHash);
      
      if (existingFile) {
        res.json({
          success: true,
          data: {
            url: existingFile.url,
            exists: true
          }
        });
      } else {
        res.json({
          success: true,
          data: {
            exists: false
          }
        });
      }
    } catch (error) {
      res.status(500).json({
        success: false,
        message: '检查文件失败',
        error: error.message
      });
    }
  };
  
  /**
   * 上传分片
   */
  public uploadChunk = async (req: Request, res: Response): Promise<void> => {
    try {
      const { chunkIndex, totalChunks, fileName, fileHash } = req.body;
      const chunk = req.file;
      
      if (!chunk) {
        return res.status(400).json({
          success: false,
          message: '分片文件不能为空'
        });
      }
      
      // 创建临时目录
      const chunkDir = path.join(this.tempDir, fileHash);
      await fs.mkdir(chunkDir, { recursive: true });
      
      // 保存分片文件
      const chunkPath = path.join(chunkDir, `chunk_${chunkIndex}`);
      await fs.writeFile(chunkPath, chunk.buffer);
      
      res.json({
        success: true,
        message: '分片上传成功',
        data: {
          chunkIndex: parseInt(chunkIndex),
          totalChunks: parseInt(totalChunks)
        }
      });
    } catch (error) {
      res.status(500).json({
        success: false,
        message: '分片上传失败',
        error: error.message
      });
    }
  };
  
  /**
   * 合并分片
   */
  public mergeChunks = async (req: Request, res: Response): Promise<void> => {
    try {
      const { fileHash, fileName, totalChunks } = req.body;
      const chunkDir = path.join(this.tempDir, fileHash);
      
      // 检查所有分片是否都已上传
      const chunkFiles = [];
      for (let i = 0; i < totalChunks; i++) {
        const chunkPath = path.join(chunkDir, `chunk_${i}`);
        try {
          await fs.access(chunkPath);
          chunkFiles.push(chunkPath);
        } catch (error) {
          return res.status(400).json({
            success: false,
            message: `分片 ${i} 不存在`
          });
        }
      }
      
      // 生成最终文件路径
      const fileExtension = path.extname(fileName);
      const finalFileName = `${uuidv4()}${fileExtension}`;
      const finalFilePath = path.join(this.uploadDir, finalFileName);
      
      // 合并分片
      const writeStream = await fs.open(finalFilePath, 'w');
      
      for (let i = 0; i < totalChunks; i++) {
        const chunkPath = path.join(chunkDir, `chunk_${i}`);
        const chunkData = await fs.readFile(chunkPath);
        await writeStream.write(chunkData);
      }
      
      await writeStream.close();
      
      // 清理临时文件
      await fs.rmdir(chunkDir, { recursive: true });
      
      // 保存文件信息到数据库
      const fileRecord = await this.fileService.create({
        originalName: fileName,
        fileName: finalFileName,
        filePath: finalFilePath,
        fileHash,
        fileSize: (await fs.stat(finalFilePath)).size,
        mimeType: this.getMimeType(fileExtension),
        uploadedBy: req.user.id
      });
      
      res.json({
        success: true,
        message: '文件上传成功',
        data: {
          id: fileRecord.id,
          url: `/uploads/${finalFileName}`,
          originalName: fileName,
          size: fileRecord.fileSize
        }
      });
    } catch (error) {
      res.status(500).json({
        success: false,
        message: '合并分片失败',
        error: error.message
      });
    }
  };
  
  /**
   * 确保目录存在
   */
  private async ensureDirectories(): Promise<void> {
    await fs.mkdir(this.uploadDir, { recursive: true });
    await fs.mkdir(this.tempDir, { recursive: true });
  }
  
  /**
   * 获取MIME类型
   */
  private getMimeType(extension: string): string {
    const mimeTypes: Record<string, string> = {
      '.jpg': 'image/jpeg',
      '.jpeg': 'image/jpeg',
      '.png': 'image/png',
      '.gif': 'image/gif',
      '.webp': 'image/webp',
      '.mp4': 'video/mp4',
      '.avi': 'video/avi',
      '.mov': 'video/quicktime',
      '.wmv': 'video/x-ms-wmv',
      '.mp3': 'audio/mpeg',
      '.wav': 'audio/wav',
      '.aac': 'audio/aac',
      '.flac': 'audio/flac',
      '.pdf': 'application/pdf',
      '.doc': 'application/msword',
      '.docx': 'application/vnd.openxmlformats-officedocument.wordprocessingml.document'
    };
    
    return mimeTypes[extension.toLowerCase()] || 'application/octet-stream';
  }
}
```

## 🎨 React上传组件

### 拖拽上传组件
```typescript
import React, { useState, useRef, DragEvent } from 'react';
import { Upload, Button, Progress, message } from 'antd';
import { UploadOutlined, InboxOutlined } from '@ant-design/icons';
import { ChunkUploader } from '@/utils/ChunkUploader';

interface FileUploadProps {
  accept?: string;
  maxSize?: number;
  multiple?: boolean;
  onSuccess?: (files: UploadedFile[]) => void;
  onError?: (error: string) => void;
}

interface UploadedFile {
  id: string;
  url: string;
  name: string;
  size: number;
}

interface UploadProgress {
  [key: string]: number;
}

const FileUpload: React.FC<FileUploadProps> = ({
  accept = 'image/*,video/*,audio/*',
  maxSize = 100 * 1024 * 1024, // 100MB
  multiple = true,
  onSuccess,
  onError
}) => {
  const [uploading, setUploading] = useState(false);
  const [progress, setProgress] = useState<UploadProgress>({});
  const [dragOver, setDragOver] = useState(false);
  const fileInputRef = useRef<HTMLInputElement>(null);
  const uploader = new ChunkUploader();

  /**
   * 处理文件选择
   */
  const handleFileSelect = (files: FileList | null) => {
    if (!files || files.length === 0) return;

    const fileArray = Array.from(files);
    const validFiles = fileArray.filter(file => {
      if (file.size > maxSize) {
        message.error(`文件 ${file.name} 超过大小限制`);
        return false;
      }
      return true;
    });

    if (validFiles.length > 0) {
      uploadFiles(validFiles);
    }
  };

  /**
   * 上传文件
   */
  const uploadFiles = async (files: File[]) => {
    setUploading(true);
    const uploadedFiles: UploadedFile[] = [];
    const newProgress: UploadProgress = {};

    try {
      for (const file of files) {
        const fileId = `${file.name}_${Date.now()}`;
        newProgress[fileId] = 0;
        setProgress({ ...newProgress });

        try {
          const url = await uploader.uploadFile(file, (progressValue) => {
            setProgress(prev => ({
              ...prev,
              [fileId]: progressValue
            }));
          });

          uploadedFiles.push({
            id: fileId,
            url,
            name: file.name,
            size: file.size
          });

          // 完成后移除进度
          setProgress(prev => {
            const updated = { ...prev };
            delete updated[fileId];
            return updated;
          });

        } catch (error) {
          message.error(`文件 ${file.name} 上传失败: ${error.message}`);
          onError?.(error.message);
        }
      }

      if (uploadedFiles.length > 0) {
        message.success(`成功上传 ${uploadedFiles.length} 个文件`);
        onSuccess?.(uploadedFiles);
      }

    } finally {
      setUploading(false);
      setProgress({});
    }
  };

  /**
   * 拖拽事件处理
   */
  const handleDragOver = (e: DragEvent) => {
    e.preventDefault();
    setDragOver(true);
  };

  const handleDragLeave = (e: DragEvent) => {
    e.preventDefault();
    setDragOver(false);
  };

  const handleDrop = (e: DragEvent) => {
    e.preventDefault();
    setDragOver(false);
    handleFileSelect(e.dataTransfer.files);
  };

  /**
   * 点击上传
   */
  const handleClick = () => {
    fileInputRef.current?.click();
  };

  return (
    <div className="file-upload">
      {/* 隐藏的文件输入 */}
      <input
        ref={fileInputRef}
        type="file"
        accept={accept}
        multiple={multiple}
        style={{ display: 'none' }}
        onChange={(e) => handleFileSelect(e.target.files)}
      />

      {/* 拖拽上传区域 */}
      <div
        className={`upload-dragger ${dragOver ? 'drag-over' : ''}`}
        onDragOver={handleDragOver}
        onDragLeave={handleDragLeave}
        onDrop={handleDrop}
        onClick={handleClick}
        style={{
          border: '2px dashed #d9d9d9',
          borderRadius: '6px',
          background: dragOver ? '#fafafa' : '#fff',
          padding: '40px 20px',
          textAlign: 'center',
          cursor: 'pointer',
          transition: 'all 0.3s'
        }}
      >
        <InboxOutlined style={{ fontSize: '48px', color: '#999' }} />
        <div style={{ marginTop: '16px' }}>
          <p style={{ fontSize: '16px', color: '#666' }}>
            点击或拖拽文件到此区域上传
          </p>
          <p style={{ fontSize: '14px', color: '#999' }}>
            支持单个或批量上传，最大文件大小 {Math.round(maxSize / 1024 / 1024)}MB
          </p>
        </div>
      </div>

      {/* 上传进度 */}
      {Object.keys(progress).length > 0 && (
        <div style={{ marginTop: '16px' }}>
          {Object.entries(progress).map(([fileId, progressValue]) => (
            <div key={fileId} style={{ marginBottom: '8px' }}>
              <div style={{ marginBottom: '4px', fontSize: '14px' }}>
                {fileId.split('_')[0]}
              </div>
              <Progress
                percent={Math.round(progressValue)}
                status={progressValue === 100 ? 'success' : 'active'}
                strokeColor={{
                  '0%': '#108ee9',
                  '100%': '#87d068'
                }}
              />
            </div>
          ))}
        </div>
      )}

      {/* 上传按钮 */}
      <div style={{ marginTop: '16px', textAlign: 'center' }}>
        <Button
          type="primary"
          icon={<UploadOutlined />}
          loading={uploading}
          onClick={handleClick}
        >
          {uploading ? '上传中...' : '选择文件'}
        </Button>
      </div>
    </div>
  );
};

export default FileUpload;
```

```typescript
import React, { useState } from 'react';
```

```typescript
/**
 * 图片压缩工具
 */
export class ImageCompressor {
  /**
   * 压缩图片
   */
  static async compress(
    file: File,
    options: {
      maxWidth?: number;
      maxHeight?: number;
      quality?: number;
      format?: 'jpeg' | 'png' | 'webp';
    } = {}
  ): Promise<File> {
    const {
      maxWidth = 1920,
      maxHeight = 1080,
      quality = 0.8,
      format = 'jpeg'
    } = options;

    return new Promise((resolve, reject) => {
      const canvas = document.createElement('canvas');
      const ctx = canvas.getContext('2d');
      const img = new Image();

      img.onload = () => {
        // 计算新尺寸
        let { width, height } = img;
        
        if (width > maxWidth) {
          height = (height * maxWidth) / width;
          width = maxWidth;
        }
        
        if (height > maxHeight) {
          width = (width * maxHeight) / height;
          height = maxHeight;
        }

        // 设置画布尺寸
        canvas.width = width;
        canvas.height = height;

        // 绘制图片
        ctx?.drawImage(img, 0, 0, width, height);

        // 转换为Blob
        canvas.toBlob(
          (blob) => {
            if (blob) {
              const compressedFile = new File([blob], file.name, {
                type: `image/${format}`,
                lastModified: Date.now()
              });
              resolve(compressedFile);
            } else {
              reject(new Error('图片压缩失败'));
            }
          },
          `image/${format}`,
          quality
        );
      };

      img.onerror = () => reject(new Error('图片加载失败'));
      img.src = URL.createObjectURL(file);
    });
  }
}
```

```typescript
/**
 * 视频帧提取工具
 */
export class VideoFrameExtractor {
  /**
   * 提取视频封面
   */
  static async extractCover(file: File, time: number = 1): Promise<File> {
    return new Promise((resolve, reject) => {
      const video = document.createElement('video');
      const canvas = document.createElement('canvas');
      const ctx = canvas.getContext('2d');

      video.onloadedmetadata = () => {
        canvas.width = video.videoWidth;
        canvas.height = video.videoHeight;
        video.currentTime = time;
      };

      video.onseeked = () => {
        ctx?.drawImage(video, 0, 0);
        
        canvas.toBlob((blob) => {
          if (blob) {
            const coverFile = new File([blob], `${file.name}_cover.jpg`, {
              type: 'image/jpeg',
              lastModified: Date.now()
            });
            resolve(coverFile);
          } else {
            reject(new Error('封面提取失败'));
          }
        }, 'image/jpeg', 0.8);
      };

      video.onerror = () => reject(new Error('视频加载失败'));
      video.src = URL.createObjectURL(file);
    });
  }

  /**
   * 提取多个关键帧
   */
  static async extractKeyFrames(
    file: File, 
    count: number = 5
  ): Promise<File[]> {
    return new Promise((resolve, reject) => {
      const video = document.createElement('video');
      const canvas = document.createElement('canvas');
      const ctx = canvas.getContext('2d');
      const frames: File[] = [];
      let currentFrame = 0;

      video.onloadedmetadata = () => {
        canvas.width = video.videoWidth;
        canvas.height = video.videoHeight;
        
        const interval = video.duration / count;
        video.currentTime = interval * currentFrame;
      };

      video.onseeked = () => {
        ctx?.drawImage(video, 0, 0);
        
        canvas.toBlob((blob) => {
          if (blob) {
            const frameFile = new File([blob], `${file.name}_frame_${currentFrame}.jpg`, {
              type: 'image/jpeg',
              lastModified: Date.now()
            });
            frames.push(frameFile);
          }

          currentFrame++;
          if (currentFrame < count) {
            const interval = video.duration / count;
            video.currentTime = interval * currentFrame;
          } else {
            resolve(frames);
          }
        }, 'image/jpeg', 0.8);
      };

      video.onerror = () => reject(new Error('视频加载失败'));
      video.src = URL.createObjectURL(file);
    });
  }
}
```
