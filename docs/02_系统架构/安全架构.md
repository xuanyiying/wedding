# 婚礼主持俱乐部官方网站 - 安全架构

## 🔒 安全概述

### 安全理念
- **纵深防御**：多层安全防护，确保系统安全
- **最小权限**：用户和系统组件仅获得必要权限
- **零信任架构**：不信任任何网络流量，验证所有访问
- **持续监控**：实时监控和响应安全威胁

### 安全目标
- **机密性**：保护敏感数据不被未授权访问
- **完整性**：确保数据在传输和存储过程中不被篡改
- **可用性**：保证系统服务的持续可用
- **可审计性**：记录所有安全相关操作和事件

### 敏感数据处理
```javascript
// 密码加密
const bcrypt = require('bcrypt');
const saltRounds = 12;

async function hashPassword(password) {
  return await bcrypt.hash(password, saltRounds);
}

async function verifyPassword(password, hash) {
  return await bcrypt.compare(password, hash);
}

// 敏感字段加密
const crypto = require('crypto');
const algorithm = 'aes-256-gcm';

function encrypt(text, key) {
  const iv = crypto.randomBytes(16);
  const cipher = crypto.createCipher(algorithm, key, iv);
  
  let encrypted = cipher.update(text, 'utf8', 'hex');
  encrypted += cipher.final('hex');
  
  const authTag = cipher.getAuthTag();
  
  return {
    encrypted,
    iv: iv.toString('hex'),
    authTag: authTag.toString('hex')
  };
}

function decrypt(encryptedData, key) {
  const decipher = crypto.createDecipher(
    algorithm, 
    key, 
    Buffer.from(encryptedData.iv, 'hex')
  );
  
  decipher.setAuthTag(Buffer.from(encryptedData.authTag, 'hex'));
  
  let decrypted = decipher.update(encryptedData.encrypted, 'hex', 'utf8');
  decrypted += decipher.final('utf8');
  
  return decrypted;
}

// 数据脱敏
function maskPhone(phone) {
  return phone.replace(/(\d{3})\d{4}(\d{4})/, '$1****$2');
}

function maskEmail(email) {
  const [username, domain] = email.split('@');
  const maskedUsername = username.charAt(0) + 
    '*'.repeat(username.length - 2) + 
    username.charAt(username.length - 1);
  return `${maskedUsername}@${domain}`;
}
```

```javascript
// 异常登录检测
async function detectAbnormalLogin(loginData) {
  const { userId, ip, userAgent, timestamp } = loginData;
  
  // 检查异地登录
  const lastLogin = await getLastLogin(userId);
  if (lastLogin && isDistantLocation(lastLogin.ip, ip)) {
    await createSecurityAlert({
      type: 'DISTANT_LOGIN',
      userId,
      details: { currentIp: ip, lastIp: lastLogin.ip }
    });
  }
  
  // 检查异常时间登录
  const hour = new Date(timestamp).getHours();
  if (hour < 6 || hour > 23) {
    await createSecurityAlert({
      type: 'UNUSUAL_TIME_LOGIN',
      userId,
      details: { loginTime: timestamp }
    });
  }
  
  // 检查设备指纹
  if (lastLogin && lastLogin.userAgent !== userAgent) {
    await createSecurityAlert({
      type: 'NEW_DEVICE_LOGIN',
      userId,
      details: { newDevice: userAgent }
    });
  }
}

// 暴力破解检测
const loginAttempts = new Map();

function detectBruteForce(ip, username, success) {
  const key = `${ip}:${username}`;
  const attempts = loginAttempts.get(key) || [];
  
  attempts.push({
    timestamp: Date.now(),
    success
  });
  
  // 保留最近15分钟的记录
  const fifteenMinutesAgo = Date.now() - 15 * 60 * 1000;
  const recentAttempts = attempts.filter(
    attempt => attempt.timestamp > fifteenMinutesAgo
  );
  
  loginAttempts.set(key, recentAttempts);
  
  // 检查失败次数
  const failedAttempts = recentAttempts.filter(
    attempt => !attempt.success
  );
  
  if (failedAttempts.length >= 5) {
    createSecurityAlert({
      type: 'BRUTE_FORCE_ATTACK',
      ip,
      username,
      details: { attemptCount: failedAttempts.length }
    });
    
    // 临时封禁IP
    addToBlacklist(ip, 60 * 60 * 1000); // 1小时
  }
}
```
